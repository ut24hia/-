<script>
  // Web Audio Context (Long Sustain Pure Synth)
  const audioCtx = new (window.AudioContext || window.webkitAudioContext)();

  function playFreq(freq, duration = 2.5) {
    if (audioCtx.state === 'suspended') audioCtx.resume();
    const osc = audioCtx.createOscillator();
    const gain = audioCtx.createGain();
    
    osc.type = 'triangle';
    osc.frequency.setValueAtTime(freq, audioCtx.currentTime);

    // Envelope for longer piano-like sustain
    gain.gain.setValueAtTime(0.4, audioCtx.currentTime);
    gain.gain.exponentialRampToValueAtTime(0.0001, audioCtx.currentTime + duration);

    osc.connect(gain);
    gain.connect(audioCtx.destination);
    osc.start();
    osc.stop(audioCtx.currentTime + duration);
  }

  // Note Data Definition (Strict German Names)
  const NOTES = [
    { name: "C", alt: "ツェー", freq: 261.63, isBlack: false },
    { name: "Cis / Des", alt: "チス / デス", freq: 277.18, isBlack: true, slot: 0 },
    { name: "D", alt: "デー", freq: 293.66, isBlack: false },
    { name: "Dis / Es", alt: "ディス / エス", freq: 311.13, isBlack: true, slot: 1 },
    { name: "E", alt: "エー", freq: 329.63, isBlack: false },
    { name: "F", alt: "エフ", freq: 349.23, isBlack: false },
    { name: "Fis / Ges", alt: "フィス / ゲス", freq: 369.99, isBlack: true, slot: 3 },
    { name: "G", alt: "ゲー", freq: 392.00, isBlack: false },
    { name: "Gis / As", alt: "ギス / アス", freq: 415.30, isBlack: true, slot: 4 },
    { name: "A", alt: "アー", freq: 440.00, isBlack: false },
    { name: "Ais / B", alt: "アイス / ベー", freq: 466.16, isBlack: true, slot: 5 },
    { name: "H", alt: "ハー", freq: 493.88, isBlack: false }
  ];

  // Sticker Database
  const STICKERS = [
    { id: 's1', icon: '🎸', name: 'エレキギター' },
    { id: 's2', icon: '🎹', name: 'グランドピアノ' },
    { id: 's3', icon: '🎷', name: 'サックス' },
    { id: 's4', icon: '🎺', name: 'トランペット' },
    { id: 's5', icon: '🎻', name: 'バイオリン' },
    { id: 's6', icon: '🥁', name: 'ドラム' },
    { id: 's7', icon: '🎧', name: 'ヘッドフォン' },
    { id: 's8', icon: '👑', name: '王冠' },
    { id: 's9', icon: '🌟', name: 'スター' },
    { id: 's10', icon: '🚀', name: 'ロケット' }
  ];

  // User State & Storage
  let state = {
    pts: 100,
    level: 1,
    inventory: ['s1', 's2'],
    placedStickers: [],
    mode: 'basic'
  };

  function loadState() {
    const saved = localStorage.getItem('de_music_app_state');
    if (saved) {
      try { state = JSON.parse(saved); } catch(e){}
    }
    updateUI();
  }

  function saveState() {
    localStorage.setItem('de_music_app_state', JSON.stringify(state));
    updateUI();
  }

  function updateUI() {
    document.getElementById('user-pts').textContent = state.pts;
    document.getElementById('user-level').textContent = state.level;
    renderInventory();
    renderPlacedStickers();
  }

  // Keyboard Generator
  function renderKeyboard() {
    const whiteContainer = document.getElementById('white-keys');
    const blackContainer = document.getElementById('black-keys');
    whiteContainer.innerHTML = '';
    blackContainer.innerHTML = '';

    NOTES.forEach((note) => {
      if (!note.isBlack) {
        const key = document.createElement('div');
        key.className = 'key-white';
        key.textContent = note.name;
        key.onclick = () => playFreq(note.freq);
        whiteContainer.appendChild(key);

        const slot = document.createElement('div');
        slot.className = 'key-black-slot';
        blackContainer.appendChild(slot);
      } else {
        const lastSlot = blackContainer.lastElementChild;
        if (lastSlot) {
          const bkey = document.createElement('div');
          bkey.className = 'key-black';
          bkey.textContent = note.name.split(' ')[0];
          bkey.onclick = (e) => { e.stopPropagation(); playFreq(note.freq); };
          lastSlot.appendChild(bkey);
        }
      }
    });
  }

  // Navigation Logic
  function switchTab(tabName) {
    document.querySelectorAll('nav button').forEach(b => b.classList.remove('active'));
    document.querySelectorAll('.tab-view').forEach(v => v.classList.remove('active'));
    event.target.classList.add('active');
    document.getElementById(`tab-${tabName}`).classList.add('active');
  }

  // Quiz Engine
  let currentTarget = null;

  function changeMode(mode) {
    state.mode = mode;
    nextQuestion();
  }

  function nextQuestion() {
    const msg = document.getElementById('msg-area');
    msg.textContent = '';
    const choices = document.getElementById('choices');
    choices.innerHTML = '';

    if (state.mode === 'basic') {
      const filtered = NOTES.filter(n => !n.isBlack);
      currentTarget = filtered[Math.floor(Math.random() * filtered.length)];
      filtered.forEach(n => createChoiceButton(n.name, n.name));
    } else if (state.mode === 'accidental') {
      const filtered = NOTES.filter(n => n.isBlack);
      currentTarget = filtered[Math.floor(Math.random() * filtered.length)];
      filtered.forEach(n => createChoiceButton(n.name, n.name));
    } else if (state.mode === 'dur') {
      const chords = [
        { name: "C-dur (ハ長調)", freqs: [261.63, 329.63, 392.00] },
        { name: "G-dur (ト長調)", freqs: [392.00, 493.88, 293.66] },
        { name: "F-dur (ヘ長調)", freqs: [349.23, 440.00, 261.63] }
      ];
      currentTarget = chords[Math.floor(Math.random() * chords.length)];
      chords.forEach(c => createChoiceButton(c.name, c.name));
    } else if (state.mode === 'moll') {
      const chords = [
        { name: "a-moll (イ短調)", freqs: [220.00, 261.63, 329.63] },
        { name: "e-moll (ホ短調)", freqs: [329.63, 392.00, 493.88] },
        { name: "d-moll (ニ短調)", freqs: [293.66, 349.23, 440.00] }
      ];
      currentTarget = chords[Math.floor(Math.random() * chords.length)];
      chords.forEach(c => createChoiceButton(c.name, c.name));
    } else if (state.mode === 'ear') {
      // Melody ear training with black keys
      const seq = [
        NOTES[0], // C
        NOTES[1], // Cis / Des
        NOTES[4]  // E
      ];
      currentTarget = { name: "C - Cis - E", sequence: seq };
      createChoiceButton("C - Cis - E", "C - Cis - E");
      createChoiceButton("C - D - E", "C - D - E");
      createChoiceButton("D - Dis - F", "D - Dis - F");
    }

    playCurrentQuestion();
  }

  function createChoiceButton(label, val) {
    const btn = document.createElement('button');
    btn.className = 'choice-btn';
    btn.textContent = label;
    btn.onclick = () => checkAnswer(val);
    document.getElementById('choices').appendChild(btn);
  }

  function playCurrentQuestion() {
    if (!currentTarget) return;
    if (currentTarget.freq) {
      playFreq(currentTarget.freq, 2.5);
    } else if (currentTarget.freqs) {
      currentTarget.freqs.forEach(f => playFreq(f, 2.8));
    } else if (currentTarget.sequence) {
      currentTarget.sequence.forEach((note, idx) => {
        setTimeout(() => playFreq(note.freq, 1.5), idx * 600);
      });
    }
  }

  function checkAnswer(val) {
    const msg = document.getElementById('msg-area');
    if (val === currentTarget.name) {
      msg.style.color = "#50E3C2";
      msg.textContent = "🎉 正解です！ (+20 PT)";
      state.pts += 20;
      saveState();
      setTimeout(nextQuestion, 1500);
    } else {
      msg.style.color = "#FF6B6B";
      msg.textContent = "❌ 残念！もう一度聴いてみましょう。";
    }
  }

  // Gacha System
  function drawGacha() {
    if (state.pts < 100) {
      alert("ポイントが足りません！クイズを解いてPTを貯めましょう。");
      return;
    }
    state.pts -= 100;
    const picked = STICKERS[Math.floor(Math.random() * STICKERS.length)];
    state.inventory.push(picked.id);
    saveState();

    document.getElementById('gacha-ball').textContent = picked.icon;
    document.getElementById('gacha-result').innerHTML = `<h3>【${picked.name}】を獲得！</h3>`;
  }

  // Sticker Canvas & Dragging Logic
  let selectedStickerIndex = null;

  function renderInventory() {
    const tray = document.getElementById('inventory-tray');
    tray.innerHTML = '';
    state.inventory.forEach((stId) => {
      const st = STICKERS.find(s => s.id === stId);
      if (!st) return;
      const el = document.createElement('div');
      el.className = 'inventory-item';
      el.textContent = st.icon;
      el.onclick = () => placeStickerOnCanvas(st);
      tray.appendChild(el);
    });
  }

  function placeStickerOnCanvas(stickerObj) {
    state.placedStickers.push({
      icon: stickerObj.icon,
      x: 50,
      y: 50
    });
    saveState();
  }

  function renderPlacedStickers() {
    const canvas = document.getElementById('album-canvas');
    canvas.querySelectorAll('.placed-sticker').forEach(e => e.remove());

    state.placedStickers.forEach((st, idx) => {
      const el = document.createElement('div');
      el.className = `placed-sticker ${selectedStickerIndex === idx ? 'selected' : ''}`;
      el.textContent = st.icon;
      el.style.left = `${st.x}px`;
      el.style.top = `${st.y}px`;

      // Touch / Mouse Dragging
      let isDragging = false;
      let startX, startY;

      const startMove = (clientX, clientY) => {
        isDragging = true;
        selectedStickerIndex = idx;
        startX = clientX - st.x;
        startY = clientY - st.y;
        renderPlacedStickers();
      };

      const doMove = (clientX, clientY) => {
        if (!isDragging) return;
        st.x = clientX - startX;
        st.y = clientY - startY;
        el.style.left = `${st.x}px`;
        el.style.top = `${st.y}px`;
      };

      const endMove = () => {
        if (isDragging) {
          isDragging = false;
          saveState();
        }
      };

      el.addEventListener('mousedown', (e) => startMove(e.clientX, e.clientY));
      window.addEventListener('mousemove', (e) => doMove(e.clientX, e.clientY));
      window.addEventListener('mouseup', endMove);

      el.addEventListener('touchstart', (e) => startMove(e.touches[0].clientX, e.touches[0].clientY));
      window.addEventListener('touchmove', (e) => doMove(e.touches[0].clientX, e.touches[0].clientY));
      window.addEventListener('touchend', endMove);

      canvas.appendChild(el);
    });
  }

  function removeSelectedSticker() {
    if (selectedStickerIndex !== null) {
      state.placedStickers.splice(selectedStickerIndex, 1);
      selectedStickerIndex = null;
      saveState();
    }
  }

  function setAlbumTheme(theme) {
    const canvas = document.getElementById('album-canvas');
    canvas.className = `album-canvas ${theme}`;
  }

  // Init
  window.onload = () => {
    renderKeyboard();
    loadState();
    nextQuestion();
  };
</script>
